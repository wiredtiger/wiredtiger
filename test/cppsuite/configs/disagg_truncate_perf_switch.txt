# Configuration for disagg_truncate_perf across a role change.
# The oplog is loaded as a leader, the connection then steps down in place, appends to its ingest
# table and trims. Nothing is adopted and the cache stays warm, so this measures the step down and
# the truncates after it.
# The insert volume has to fit the cache: a follower's ingest table is in memory and nothing
# collects it here, since a stepped down connection adopts no checkpoints.

insert_mb=256,
marker_size_mb=8,
oplog_size_mb=512,
role=switch,
value_size=1000,
metrics_monitor=(enabled=false),
operation_tracker=(enabled=false),
workload_manager=(
    populate_config=(thread_count=4),
    insert_config=(thread_count=4),
    custom_config=(thread_count=1),
    checkpoint_config=(thread_count=1,op_rate=2s))
